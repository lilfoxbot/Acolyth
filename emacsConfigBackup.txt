(let ((raylib-bin "C:/raylib/w64devkit/bin"))
  (when (file-directory-p raylib-bin)

    (setenv "PATH" (concat (replace-regexp-in-string "/" "\\\\" raylib-bin) ";" (getenv "PATH")))))

(defun raylib-build-and-run ()
  "Save all buffers, compile the current Raylib C file, and run it."
  (interactive)
  (save-some-buffers t)
  (let* ((src (file-name-nondirectory (buffer-file-name)))
         (exe (concat (file-name-sans-extension src) ".exe"))
         (cmd (format "gcc %s -o %s -O1 -Wall -std=c99 -Wno-missing-braces -I C:/raylib/raylib/src -L C:/raylib/raylib/src -lraylib -lopengl32 -lgdi32 -lwinmm && %s"
                      src exe exe)))
    (compile cmd)))

(with-eval-after-load 'cc-mode
  (define-key c-mode-base-map (kbd "<f5>") 'raylib-build-and-run))
;;

;; BUILD/DEBUG
(defun raylib-build-and-debug ()
  "Save all buffers, compile the current Raylib C file with debug symbols, and launch GDB."
  (interactive)
  (save-some-buffers t)
  (let* ((dir default-directory)
         (src (file-name-nondirectory (buffer-file-name)))
         (exe (concat (file-name-sans-extension src) ".exe"))
         (cmd (format "gcc %s -o %s -g -O0 -Wall -std=c99 -Wno-missing-braces -I C:/raylib/raylib/src -L C:/raylib/raylib/src -lraylib -lopengl32 -lgdi32 -lwinmm"
                      src exe))
         (comp-buf (compile cmd)))
    (with-current-buffer comp-buf
      (setq-local compilation-finish-functions
                  (list `(lambda (buf status)
                           (if (string-match-p "finished" status)
                               (let ((default-directory ,dir))
                                 (gdb ,(format "gdb -i=mi %s" exe))
                                 (gdb-many-windows 1))
                             (message "Build failed; aborting debug."))))))))

(with-eval-after-load 'cc-mode
  (define-key c-mode-base-map (kbd "<f6>") #'raylib-build-and-debug))
