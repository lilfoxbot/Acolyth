;; HOTKEYS
;; NOTE: DO NOT USE "C-i" (it is a "tab" override)

;; Enable C-c (copy), C-v (paste), C-x (cut), and C-z (undo) when text is selected

;; Select All
(keymap-global-set "C-a" #'mark-whole-buffer)

;; Save Buffer
(keymap-global-set "C-s" 'save-buffer)

;; Search Text
(keymap-global-set "C-f" 'isearch-forward)

;; Evaluate Line (when cursor at end of line)
(global-set-key (kbd "C-e") 'eval-last-sexp)

;; Toggle Speedbar (file explorer GUI)
(global-set-key (kbd "C-SPC") 'speedbar)

;; Open Init File (currently used)
(keymap-global-set "C-p" (lambda () (interactive) (find-file user-init-file)))

;; Open Scratch Buffer

(keymap-global-set "C-S-f" #'project-find-file)
